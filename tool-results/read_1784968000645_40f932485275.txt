    74→    .trim()
    75→    .replace(/[^a-z0-9\s-]/g, '')

    77→    .replace(/-+/g, '-')
    78→    .replace(/^-+|-+$/g, '')
    79→}

   421→
   422→export default function RegisterPage() {
   423→  const router = useRouter()
   424→  const searchParams = useSearchParams()
   425→
   426→  /* ---- State ---- */
   427→  const [plans, setPlans] = useState<Plan[]>([])
   428→  const [plansLoading, setPlansLoading] = useState(true)
   429→  const [selectedPlanId, setSelectedPlanId] = useState('')
   430→  const [showPassword, setShowPassword] = useState(false)
   431→  const [showConfirm, setShowConfirm] = useState(false)
   432→  const [submitting, setSubmitting] = useState(false)
   433→  const [success, setSuccess] = useState(false)
   434→  const [successData, setSuccessData] = useState({ companyName: '', slug: '' })
   435→  const [errors, setErrors] = useState<FieldErrors>({})
   436→
   437→  const [companyName, setCompanyName] = useState('')
   438→  const [slug, setSlug] = useState('')
   439→  const [phone, setPhone] = useState('')
   440→  const [adminName, setAdminName] = useState('')
   441→  const [adminEmail, setAdminEmail] = useState('')
   442→  const [adminPassword, setAdminPassword] = useState('')
   443→  const [confirmPassword, setConfirmPassword] = useState('')
   444→  const [termsAccepted, setTermsAccepted] = useState(false)
   445→
   446→  /* ---- Fetch plans ---- */
   447→  useEffect(() => {
   448→    async function fetchPlans() {
   449→      try {
   450→        const res = await fetch('/api/platform/plans')
   451→        if (res.ok) {
   452→          const data = await res.json()
   453→          const activePlans = (data.plans ?? []).filter(
   454→            (p: Plan) => p.active,
   455→          )
   456→          setPlans(activePlans.length > 0 ? activePlans : FALLBACK_PLANS)
   457→          const free = activePlans.find(
   458→            (p: Plan) => p.priceMonthly === 0 || p.name === 'Free Trial',
   459→          )
   460→          if (free) setSelectedPlanId(free.id)
   461→          return
   462→        }
   463→      } catch {
   464→        /* fall through */
   465→      }
   466→      // Fallback plans
   467→      setPlans(FALLBACK_PLANS)
   468→      setSelectedPlanId('free-trial')
   469→    }
   470→    fetchPlans().finally(() => setPlansLoading(false))
   471→  }, [])
   472→
   473→  /* ---- Pre-select plan from URL ---- */
   474→  useEffect(() => {
   475→    const planParam = searchParams.get('plan')
   476→    if (planParam && plans.length > 0) {
   477→      const match = plans.find(
   478→        (p) =>
   479→          p.id === planParam ||
   480→          p.name.toLowerCase() === planParam.toLowerCase(),
   481→      )
   482→      if (match) setSelectedPlanId(match.id)
   483→    }
   484→  }, [searchParams, plans])
   485→
   486→  /* ---- Auto-generate slug from company name ---- */
   487→  const handleCompanyNameChange = useCallback(
   488→    (value: string) => {
   489→      setCompanyName(value)
   490→      // Only auto-generate if slug hasn't been manually edited
   491→      const generated = generateSlug(value)
   492→      if (!slug || generateSlug(companyName) === slug) {
   493→        setSlug(generated)
   494→      }
   495→    },
   496→    [companyName, slug],
   497→  )
   498→
   499→  const handleSlugChange = useCallback((value: string) => {
   500→    setSlug(value)
   501→  }, [])
   502→
   503→  /* ---- Validation ---- */
   504→  const validate = useCallback((): boolean => {
   505→    const errs: FieldErrors = {}
   506→
   507→    if (!companyName.trim()) errs.companyName = 'Company name is required'
   508→    if (!slug.trim()) errs.slug = 'Company URL is required'
   509→    else if (!SLUG_RE.test(slug))
   510→      errs.slug = 'Only lowercase letters, numbers, and hyphens'
   511→    else if (slug.length < 2) errs.slug = 'Slug must be at least 2 characters'
   512→
   513→    if (!adminName.trim()) errs.adminName = 'Full name is required'
   514→    if (!adminEmail.trim()) errs.adminEmail = 'Email is required'
   515→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(adminEmail.trim()))
   516→      errs.adminEmail = 'Invalid email format'
   517→
   518→    if (!adminPassword) errs.adminPassword = 'Password is required'
   519→    else if (adminPassword.length < 6)
   520→      errs.adminPassword = 'Password must be at least 6 characters'
   521→
   522→    if (!confirmPassword) errs.confirmPassword = 'Please confirm your password'
   523→    else if (confirmPassword !== adminPassword)
   524→      errs.confirmPassword = 'Passwords do not match'
   525→
   526→    if (!selectedPlanId) errs.planId = 'Please select a plan'
   527→
   528→    if (!termsAccepted) errs.terms = 'You must accept the terms'
   529→
   530→    setErrors(errs)
   531→    return Object.keys(errs).length === 0
   532→  }, [
   533→    companyName,
   534→    slug,
   535→    adminName,
   536→    adminEmail,
   537→    adminPassword,
   538→    confirmPassword,
   539→    selectedPlanId,
   540→    termsAccepted,
   541→  ])
   542→
   543→  /* ---- Clear field error on change ---- */
   544→  const clearFieldError = useCallback((field: keyof FieldErrors) => {
   545→    setErrors((prev) => {
   546→      if (!prev[field]) return prev
   547→      const next = { ...prev }
   548→      delete next[field]
   549→      return next
   550→    })
   551→  }, [])
   552→
   553→  /* ---- Submit ---- */
   554→  const handleSubmit = useCallback(
   555→    async (e: React.FormEvent) => {
   556→      e.preventDefault()
   557→      if (!validate()) return
   558→
   559→      setSubmitting(true)
   560→      try {
   561→        const res = await fetch('/api/register', {
   562→          method: 'POST',
   563→          headers: { 'Content-Type': 'application/json' },
   564→          body: JSON.stringify({
   565→            companyName: companyName.trim(),
   566→            slug: slug.trim(),
   567→            adminName: adminName.trim(),
   568→            adminEmail: adminEmail.trim().toLowerCase(),
   569→            adminPassword,
   570→            phone: phone.trim() || undefined,
   571→            planId: selectedPlanId,
   572→          }),
   573→        })
   574→        const data = await res.json()
   575→
   576→        if (!res.ok) {
   577→          toast.error(data.error ?? 'Registration failed. Please try again.')
   578→          if (data.error?.toLowerCase().includes('slug') || data.error?.toLowerCase().includes('url') || data.error?.toLowerCase().includes('taken')) {
   579→            setErrors((prev) => ({ ...prev, slug: data.error }))
   580→          }
   581→          return
   582→        }
   583→
   584→        // Success
   585→        setSuccessData({
   586→          companyName: data.tenant?.name ?? companyName.trim(),
   587→          slug: data.tenant?.slug ?? slug.trim(),
   588→        })
   589→        setSuccess(true)
   590→        toast.success('Workspace created successfully!')
   591→      } catch {
   592→        toast.error('Network error. Please check your connection and try again.')
   593→      } finally {
   594→        setSubmitting(false)
   595→      }
   596→    },
   597→    [
   598→      validate,
   599→      companyName,
