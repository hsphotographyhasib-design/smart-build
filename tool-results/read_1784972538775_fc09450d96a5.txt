    42→function SectionFadeIn({
    43→  children,
    44→  className,
    45→  delay = 0,
    46→}: {
    47→  children: React.ReactNode
    48→  className?: string
    49→  delay?: number
    50→}) {
    51→  return (
    52→    <motion.div
    53→      initial={{ opacity: 0, y: 20 }}
    54→      whileInView={{ opacity: 1, y: 0 }}
    55→      viewport={{ once: true, margin: '-80px' }}
    56→      transition={{ duration: 0.5, delay, ease: [0.22, 1, 0.36, 1] }}
    57→      className={className}
    58→    >
    59→      {children}
    60→    </motion.div>
    61→  )
    62→}

    64→function StaggerContainer({
    65→  children,
    66→  className,
    67→}: {
    68→  children: React.ReactNode
    69→  className?: string
    70→}) {
    71→  return (
    72→    <motion.div
    73→      initial="hidden"
    74→      whileInView="show"
    75→      viewport={{ once: true, margin: '-60px' }}
    76→      variants={{
    77→        hidden: {},
    78→        show: { transition: { staggerChildren: 0.08 } },
    79→      }}
    80→      className={className}
    81→    >
    82→      {children}
    83→    </motion.div>
    84→  )
    85→}
