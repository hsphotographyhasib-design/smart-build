    94→function formatStorage(mb: number): string {
    95→  if (mb >= 1000000) return `${(mb / 1000000).toFixed(0)} TB`
    96→  if (mb >= 1000) return `${(mb / 1000).toFixed(0)} GB`
    97→  return `${mb} MB`
    98→}
    99→
   100→function formatNumber(n: number): string {
   101→  if (n >= 1000000) return `${(n / 1000000).toFixed(0)}M`
   102→  if (n >= 1000) return `${(n / 1000).toFixed(0)}K`
   103→  return n === 0 ? '0' : n.toLocaleString()
   104→}
   105→
   106→function formatPrice(price: number): string {
   107→  if (price === 0) return 'Free'
   108→  return `$${price.toLocaleString()}`
   109→}
   110→
   111→function getSavings(monthly: number, annual: number): number {
   112→  if (monthly === 0 || annual === 0) return 0
   113→  const annualTotal = annual * 12
   114→  const monthlyTotal = monthly * 12
   115→  return Math.round(((monthlyTotal - annualTotal) / monthlyTotal) * 100)
   116→}

   118→function getPlanIconKey(name: string): string {
   119→  const lower = name.toLowerCase()
   120→  if (lower.includes('free') || lower.includes('trial')) return 'zap'
   121→  if (lower.includes('starter')) return 'shield'
   122→  if (lower.includes('professional') || lower.includes('pro')) return 'crown'
   123→  if (lower.includes('enterprise')) return 'building'
   124→  return 'users'
   125→}

   457→function formatCellValue(plan: Plan, row: (typeof COMPARISON_ROWS)[number]) {
   458→  if (row.type === 'boolean') {
   459→    const val = plan[row.key as keyof Plan] as boolean
   460→    return val ? (
   461→      <Check className="size-5 text-emerald-500 mx-auto" />
   462→    ) : (
   463→      <X className="size-5 text-gray-300 mx-auto" />
   464→    )
   465→  }
   466→  if (row.type === 'storage') {
   467→    const val = plan[row.key as keyof Plan] as number
   468→    return <span className="font-body text-sm text-gray-700">{val === 0 ? 'Unlimited' : formatStorage(val)}</span>
   469→  }
   470→  if (row.key === 'maxApiCalls') {
   471→    const val = plan[row.key as keyof Plan] as number
   472→    return <span className="font-body text-sm text-gray-700">{val === 0 ? 'Unlimited' : formatNumber(val)}</span>
   473→  }
   474→  const val = plan[row.key as keyof Plan] as number
   475→  return <span className="font-body text-sm text-gray-700">{val === 0 ? 'Unlimited' : val.toLocaleString()}</span>
   476→}
