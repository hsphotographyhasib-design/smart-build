    97→const PLAN_ICONS: Record<string, React.ElementType> = {
    98→  zap: Zap,

   105→function getPlanIconKey(name: string): string {
   106→  const lower = name.toLowerCase()
   107→  if (lower.includes('free') || lower.includes('trial')) return 'zap'
   108→  if (lower.includes('starter')) return 'shield'
   109→  if (lower.includes('professional') || lower.includes('pro')) return 'crown'
   110→  if (lower.includes('enterprise') || lower.includes('custom')) return 'building'
   111→  return 'users'
   112→}
   113→
   114→function PlanIcon({ name }: { name: string }) {
   115→  const key = getPlanIconKey(name)
   116→  const Icon = PLAN_ICONS[key] || Users
   117→  return <Icon className="h-4 w-4 text-primary" />
   118→}

   123→function formatStorage(mb: number): string {
   124→  if (mb >= 100000) return `${(mb / 1000).toFixed(0)} GB`
   125→  if (mb >= 1000) return `${(mb / 1000).toFixed(0)} GB`
   126→  return `${mb} MB`
   127→}
   128→
   129→function formatApiCalls(n: number): string {
   130→  if (n >= 1000000) return `${(n / 1000000).toFixed(0)}M`
   131→  if (n >= 1000) return `${(n / 1000).toFixed(0)}K`
   132→  return n === 0 ? '-' : String(n)
   133→}
   134→
   135→function getSavings(monthly: number, annual: number): number {
   136→  if (monthly === 0 || annual === 0) return 0
   137→  const annualTotal = annual * 12
   138→  const monthlyTotal = monthly * 12
   139→  return Math.round(((monthlyTotal - annualTotal) / monthlyTotal) * 100)
   140→}
   141→
   142→function isProfessionalPlan(name: string): boolean {
   143→  return name.toLowerCase().includes('professional')
   144→}
   145→
   146→function isEnterprisePlan(name: string): boolean {
   147→  return name.toLowerCase().includes('enterprise') || name.toLowerCase().includes('custom')
   148→}

   150→function parseFeatures(featuresStr: string): string[] {

   153→    if (Array.isArray(parsed)) return parsed
   154→    return []
   155→  } catch {
   156→    return []
   157→  }
   158→}
