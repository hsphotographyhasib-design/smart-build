   237→function getPlanIconKey(name: string): string {
   238→  const lower = name.toLowerCase()
   239→  if (lower.includes('free') || lower.includes('trial')) return 'zap'
   240→  if (lower.includes('starter')) return 'shield'
   241→  if (lower.includes('professional') || lower.includes('pro')) return 'crown'
   242→  if (lower.includes('enterprise') || lower.includes('custom')) return 'building'
   243→  return 'users'
   244→}
   245→
   246→function PlanIcon({ name }: { name: string }) {
   247→  const key = getPlanIconKey(name)
   248→  const Icon = PLAN_ICONS[key] || Users
   249→  return <Icon className="h-4 w-4 text-primary" />
   250→}

   254→/* ------------------------------------------------------------------ */
   255→function formatStorage(mb: number): string {
   256→  if (mb >= 1000000) return `${(mb / 1000 / 1000).toFixed(0)} TB`
   257→  if (mb >= 1000) return `${(mb / 1000).toFixed(0)} GB`
   258→  return `${mb} MB`
   259→}
   260→
   261→function formatApiCalls(n: number): string {
   262→  if (n >= 1000000) return `${(n / 1000000).toFixed(0)}M`
   263→  if (n >= 1000) return `${(n / 1000).toFixed(0)}K`
   264→  return n === 0 ? '-' : String(n)
   265→}
   266→
   267→function getSavings(monthly: number, annual: number): number {
   268→  if (monthly === 0 || annual === 0) return 0
   269→  const annualTotal = annual * 12
   270→  const monthlyTotal = monthly * 12
   271→  return Math.round(((monthlyTotal - annualTotal) / monthlyTotal) * 100)
   272→}
