    96→function formatStorage(mb: number): string {
    97→  if (mb >= 100000) return `${(mb / 1000).toFixed(0)} GB`
    98→  if (mb >= 1000) return `${(mb / 1000).toFixed(0)} GB`
    99→  return `${mb} MB`
   100→}
   101→
   102→function formatApiCalls(n: number): string {
   103→  if (n >= 1000000) return `${(n / 1000000).toFixed(0)}M`
   104→  if (n >= 1000) return `${(n / 1000).toFixed(0)}K`
   105→  return n === 0 ? '—' : String(n)
   106→}
   107→
   108→function getSavings(monthly: number, annual: number): number {
   109→  if (monthly === 0 || annual === 0) return 0
   110→  const annualTotal = annual * 12
   111→  const monthlyTotal = monthly * 12
   112→  return Math.round(((monthlyTotal - annualTotal) / monthlyTotal) * 100)
   113→}

   115→function isProfessionalPlan(name: string): boolean {

   117→}
   118→
   119→function isEnterprisePlan(name: string): boolean {
   120→  return name.toLowerCase().includes('enterprise') || name.toLowerCase().includes('custom')

   123→function getPlanIcon(name: string) {
   124→  const lower = name.toLowerCase()
   125→  if (lower.includes('free') || lower.includes('trial')) return Zap
   126→  if (lower.includes('starter')) return Shield
   127→  if (lower.includes('professional') || lower.includes('pro')) return Crown
   128→  if (lower.includes('enterprise') || lower.includes('custom')) return Building2
   129→  return Users
   130→}

   132→function parseFeatures(featuresStr: string): string[] {

   136→    return []
   137→  } catch {
   138→    return []
   139→  }
