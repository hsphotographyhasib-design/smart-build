   237→   237→function getPlanIconKey(name: string): string {
   238→   238→  const lower = name.toLowerCase()
   239→   239→  if (lower.includes('free') || lower.includes('trial')) return 'zap'
   240→   240→  if (lower.includes('starter')) return 'shield'
   241→   241→  if (lower.includes('professional') || lower.includes('pro')) return 'crown'
   242→   242→  if (lower.includes('enterprise') || lower.includes('custom')) return 'building'
   243→   243→  return 'users'
   244→   244→}
   245→   245→
   246→   246→function PlanIcon({ name }: { name: string }) {
   247→   247→  const key = getPlanIconKey(name)

   255→   255→function formatStorage(mb: number): string {
   256→   256→  if (mb >= 1000000) return `${(mb / 1000 / 1000).toFixed(0)} TB`
   257→   257→  if (mb >= 1000) return `${(mb / 1000).toFixed(0)} GB`
   258→   258→  return `${mb} MB`
   259→   259→}
   260→   260→
   261→   261→function formatApiCalls(n: number): string {
   262→   262→  if (n >= 1000000) return `${(n / 1000000).toFixed(0)}M`
   263→   263→  if (n >= 1000) return `${(n / 1000).toFixed(0)}K`
   264→   264→  return n === 0 ? '-' : String(n)
   265→   265→}
   266→   266→
   267→   267→function getSavings(monthly: number, annual: number): number {
   268→   268→  if (monthly === 0 || annual === 0) return 0
   269→   269→  const annualTotal = annual * 12
   270→   270→  const monthlyTotal = monthly * 12
   271→   271→  return Math.round(((monthlyTotal - annualTotal) / monthlyTotal) * 100)
   272→   272→}
