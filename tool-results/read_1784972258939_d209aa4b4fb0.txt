   128→   128→function SectionFadeIn({
   129→   129→  children,
   130→   130→  className,
   131→   131→  delay = 0,
   132→   132→}: {

   135→   135→  delay?: number
   136→   136→}) {
   137→   137→  return (

   139→   139→      initial={{ opacity: 0, y: 10 }}

   143→   143→    >
   144→   144→      {children}
   145→   145→    </motion.div>
   146→   146→  )
   147→   147→}

   153→   153→function slugify(name: string): string {
   154→   154→  return name
   155→   155→    .toLowerCase()
   156→   156→    .trim()
   157→   157→    .replace(/[^\w\s-]/g, '')
   158→   158→    .replace(/[\s_]+/g, '-')
   159→   159→    .replace(/-+/g, '-')
   160→   160→    .replace(/^-+|-+$/g, '')
   161→   161→}
