   153→function slugify(name: string): string {
   154→  return name
   155→    .toLowerCase()
   156→    .trim()
   157→    .replace(/[^\w\s-]/g, '')
   158→    .replace(/[\s_]+/g, '-')
   159→    .replace(/-+/g, '-')
   160→    .replace(/^-+|-+$/g, '')
   161→}
